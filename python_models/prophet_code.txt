import pandas as pd
import numpy as np
from prophet_model import run_prophet_forecast


def run_prophet_forecast_flexible(input_data, forecast_steps=14, column_name=None):
    try:
        # ================= INPUT HANDLING =================
        if isinstance(input_data, pd.DataFrame):
            df = input_data.copy()
        else:
            return {
                "status": "error",
                "message": "Input must be a pandas DataFrame"
            }

        # ================= AUTO COLUMN DETECTION =================
        cols = df.columns.tolist()

        # Detect date column
        if 'ds' not in cols:
            for c in cols:
                if 'date' in c.lower() or 'time' in c.lower():
                    df.rename(columns={c: 'ds'}, inplace=True)
                    break

        # Detect target column
        if 'y' not in df.columns:
            if column_name and column_name in df.columns:
                df.rename(columns={column_name: 'y'}, inplace=True)
            else:
                numeric_cols = df.select_dtypes(include=[np.number]).columns.tolist()
                if not numeric_cols:
                    return {
                        "status": "error",
                        "message": "No numeric column found for forecasting"
                    }
                df.rename(columns={numeric_cols[0]: 'y'}, inplace=True)

        # ================= VALIDATION =================
        if 'ds' not in df.columns or 'y' not in df.columns:
            return {
                "status": "error",
                "message": "Could not detect required columns (ds, y)"
            }

        df = df[['ds', 'y']].dropna()

        # ================= DATE FIX =================
        df['ds'] = pd.to_datetime(df['ds'], errors='coerce')
        df = df.dropna(subset=['ds'])

        # ================= SORT =================
        df = df.sort_values('ds')

        # ================= REMOVE DUPLICATES =================
        df = df.drop_duplicates(subset=['ds'])

        # ================= OUTLIER CLIPPING (IMPORTANT) =================
        q_low = df['y'].quantile(0.01)
        q_high = df['y'].quantile(0.99)
        df['y'] = np.clip(df['y'], q_low, q_high)

        # ================= DATA REDUCTION =================
        df = df.tail(4000)   # ⚡ faster than 5000

        if len(df) < 20:
            return {
                "status": "error",
                "message": "Not enough data after cleaning"
            }

        # ================= NORMALIZATION (KEY FIX) =================
        # Helps Prophet learn trend properly
        mean = df['y'].mean()
        std = df['y'].std() if df['y'].std() != 0 else 1
        df['y'] = (df['y'] - mean) / std

        # ================= RUN MODEL =================
        result = run_prophet_forecast(df, forecast_steps)

        # ================= DENORMALIZE OUTPUT =================
        if "forecast" in result:
            for item in result["forecast"]:
                item["value"] = float(item["value"] * std + mean)

        return result

    except Exception as e:
        return {
            "status": "error",
            "message": str(e)
        }
