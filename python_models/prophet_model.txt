import pandas as pd
import numpy as np
from prophet import Prophet
from sklearn.metrics import mean_absolute_error, mean_squared_error


def run_prophet_forecast(df, forecast_steps=7):
    try:
        df = df.copy()

        # ================= VALIDATION =================
        if 'ds' not in df.columns or 'y' not in df.columns:
            return {"status": "error", "message": "Data must contain ds and y"}

        df['ds'] = pd.to_datetime(df['ds'], errors='coerce')
        df = df.dropna(subset=['ds', 'y']).sort_values('ds')

        if len(df) < 30:
            return {"status": "error", "message": "Not enough data"}

        # ================= TRAIN / TEST SPLIT =================
        split = int(len(df) * 0.8)
        train = df.iloc[:split]
        test = df.iloc[split:]

        # ================= MODEL =================
        model = Prophet(
            daily_seasonality=True,
            weekly_seasonality=True,
            yearly_seasonality=False,
            changepoint_prior_scale=0.5   # 🔥 helps improve flexibility
        )

        model.fit(train)

        # ================= FUTURE FORECAST =================
        future = model.make_future_dataframe(periods=forecast_steps)
        forecast_df = model.predict(future)

        # ================= TEST PREDICTION (FIXED) =================
        test_forecast = model.predict(test[['ds']])
        y_true = test['y'].values
        y_pred = test_forecast['yhat'].values

        # ================= METRICS =================
        mae = mean_absolute_error(y_true, y_pred)
        rmse = np.sqrt(mean_squared_error(y_true, y_pred))

        # Safe MAPE
        mask = y_true > 10
        mape = (
            np.mean(np.abs((y_true[mask] - y_pred[mask]) / y_true[mask])) * 100
            if np.any(mask) else 0
        )

        ss_res = np.sum((y_true - y_pred) ** 2)
        ss_tot = np.sum((y_true - np.mean(y_true)) ** 2)
        r2 = 1 - (ss_res / ss_tot) if ss_tot != 0 else 0

        # ================= FORECAST OUTPUT =================
        forecast_output = []
        future_forecast = forecast_df.tail(forecast_steps)

        for i, row in enumerate(future_forecast.itertuples(), 1):
            forecast_output.append({
                "day": f"Day {i}",
                "date": row.ds.strftime("%Y-%m-%d"),
                "value": round(float(row.yhat), 2)
            })

        # ================= HISTORICAL =================
        historical = [
            {"date": row['ds'].strftime("%Y-%m-%d"), "value": float(row['y'])}
            for _, row in df.tail(20).iterrows()
        ]

        return {
            "status": "success",
            "model": "Facebook Prophet",
            "forecast": forecast_output,
            "metrics": {
                "MAE": float(mae),
                "RMSE": float(rmse),
                "MAPE": float(mape),
                "R2": float(r2)
            },
            "historical": historical,
            "data_points": len(df)
        }

    except Exception as e:
        return {"status": "error", "message": str(e)}
