import numpy as np
import pandas as pd
import json
import sys

from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score

from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense


# ================= CREATE DATA =================
def create_sequences(data, lookback):
    X, y = [], []
    for i in range(len(data) - lookback):
        X.append(data[i:i + lookback])
        y.append(data[i + lookback])
    return np.array(X), np.array(y)


# ================= MAIN FUNCTION =================
def run_lstm_forecast(df, place=None, sector=None, lookback=20, forecast_steps=7):
    try:
        if place:
            df = df[df['place'] == place]
        if sector:
            df = df[df['sector'] == sector]

        df = df.sort_values('date')
        values = df['energy_consumed'].values.reshape(-1, 1)

        # ================= SCALE =================
        scaler = MinMaxScaler()
        scaled = scaler.fit_transform(values)

        # ================= SPLIT =================
        train_size = int(len(scaled) * 0.8)
        train = scaled[:train_size]
        test = scaled[train_size:]

        # ================= SEQUENCES =================
        X_train, y_train = create_sequences(train, lookback)
        X_test, y_test = create_sequences(test, lookback)

        # ================= MODEL =================
        model = Sequential([
            LSTM(50, return_sequences=False, input_shape=(lookback, 1)),
            Dense(1)
        ])

        model.compile(optimizer='adam', loss='mse')

        model.fit(X_train, y_train, epochs=5, batch_size=32, verbose=0)

        # ================= TEST PREDICTION =================
        test_pred = model.predict(X_test, verbose=0)

        test_pred = scaler.inverse_transform(test_pred)
        y_test_actual = scaler.inverse_transform(y_test.reshape(-1, 1))

        # ================= METRICS =================
        mae = mean_absolute_error(y_test_actual, test_pred)
        rmse = np.sqrt(mean_squared_error(y_test_actual, test_pred))
        mape = np.mean(np.abs((y_test_actual - test_pred) / (y_test_actual + 1e-8))) * 100
        r2 = r2_score(y_test_actual, test_pred)

        # ================= FUTURE FORECAST =================
        last_seq = scaled[-lookback:]
        future_preds = []

        current_seq = last_seq.copy()

        for _ in range(forecast_steps):
            pred = model.predict(current_seq.reshape(1, lookback, 1), verbose=0)[0][0]
            future_preds.append(pred)

            current_seq = np.append(current_seq[1:], [[pred]], axis=0)

        future_preds = scaler.inverse_transform(np.array(future_preds).reshape(-1, 1))

        return {
            "forecast": future_preds.flatten().tolist(),
            "metrics": {
                "mae": float(mae),
                "rmse": float(rmse),
                "mape": float(mape),
                "r2_score": float(r2)
            }
        }

    except Exception as e:
        return {"error": str(e)}


# ================= NODE ENTRY =================
if __name__ == "__main__":
    try:
        input_data = json.loads(sys.stdin.read())
        values = input_data["values"]

        df = pd.DataFrame({
            "date": pd.date_range(start="2020-01-01", periods=len(values)),
            "energy_consumed": values
        })

        result = run_lstm_forecast(df)

        print(json.dumps(result))

    except Exception as e:
        print(json.dumps({"error": str(e)}))
